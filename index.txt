<script>
  const STORAGE_KEY = "yos_finance_data";
  let entries = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
  let editingId = null;

  const form = document.getElementById("entryForm");
  const tableBody = document.getElementById("tableBody");
  const filterMonth = document.getElementById("filterMonth");

  document.getElementById("date").valueAsDate = new Date();

  form.addEventListener("submit", function(e) {
    e.preventDefault();
    const entry = {
      id: editingId || Date.now(),
      type: document.getElementById("type").value,
      date: document.getElementById("date").value,
      category: document.getElementById("category").value.trim(),
      amount: parseFloat(document.getElementById("amount").value),
      recorder: document.getElementById("recorder").value.trim(),
      description: document.getElementById("description").value.trim()
    };

    if (editingId) {
      const idx = entries.findIndex(e => e.id === editingId);
      entries[idx] = entry;
      editingId = null;
      document.getElementById("submitBtn").textContent = "+ บันทึกรายการ";
    } else {
      entries.push(entry);
    }

    save();
    form.reset();
    document.getElementById("date").valueAsDate = new Date();
    render();
  });

  filterMonth.addEventListener("change", render);

  function clearFilter() {
    filterMonth.value = "";
    render();
  }

  function save() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(entries));
  }

  function editEntry(id) {
    const entry = entries.find(e => e.id === id);
    if (!entry) return;
    document.getElementById("type").value = entry.type;
    document.getElementById("date").value = entry.date;
    document.getElementById("category").value = entry.category;
    document.getElementById("amount").value = entry.amount;
    document.getElementById("recorder").value = entry.recorder;
    document.getElementById("description").value = entry.description;
    editingId = id;
    document.getElementById("submitBtn").textContent = "บันทึกการแก้ไข";
    window.scrollTo({top: 0, behavior: "smooth"});
  }

  function deleteEntry(id) {
    if (!confirm("ยืนยันการลบรายการนี้?")) return;
    entries = entries.filter(e => e.id !== id);
    save();
    render();
  }

  function clearAll() {
    if (!confirm("ยืนยันการล้างข้อมูลทั้งหมด? ไม่สามารถกู้คืนได้")) return;
    entries = [];
    save();
    render();
  }

  function formatMoney(n) {
    return n.toLocaleString("th-TH", {minimumFractionDigits: 2, maximumFractionDigits: 2});
  }

  function render() {
    let filtered = entries;
    if (filterMonth.value) {
      filtered = entries.filter(e => e.date.startsWith(filterMonth.value));
    }

    filtered.sort((a, b) => new Date(b.date) - new Date(a.date));

    tableBody.innerHTML = "";
    if (filtered.length === 0) {
      tableBody.innerHTML = '<tr><td colspan="7" class="empty-row">ยังไม่มีรายการ</td></tr>';
    } else {
      filtered.forEach(e => {
        const tr = document.createElement("tr");
        tr.innerHTML = `
          <td>${e.date}</td>
          <td class="${e.type === 'income' ? 'type-income' : 'type-expense'}">${e.type === 'income' ? 'รายรับ' : 'รายจ่าย'}</td>
          <td>${e.category}</td>
          <td>${formatMoney(e.amount)}</td>
          <td>${e.recorder}</td>
          <td>${e.description || '-'}</td>
          <td class="actions">
            <button onclick="editEntry(${e.id})">แก้ไข</button>
            <button class="btn-danger" onclick="deleteEntry(${e.id})">ลบ</button>
          </td>
        `;
        tableBody.appendChild(tr);
      });
    }

    const totalIncome = filtered.filter(e => e.type === "income").reduce((s, e) => s + e.amount, 0);
    const totalExpense = filtered.filter(e => e.type === "expense").reduce((s, e) => s + e.amount, 0);
    document.getElementById("totalIncome").textContent = formatMoney(totalIncome);
    document.getElementById("totalExpense").textContent = formatMoney(totalExpense);
    document.getElementById("balance").textContent = formatMoney(totalIncome - totalExpense);
  }

  function exportCSV() {
    if (entries.length === 0) {
      alert("ไม่มีข้อมูลให้ดาวน์โหลด");
      return;
    }
    let csv = "วันที่,ประเภท,หมวดหมู่,จำนวนเงิน,ผู้บันทึก,รายละเอียด\n";
    entries.forEach(e => {
      const type = e.type === "income" ? "รายรับ" : "รายจ่าย";
      csv += `${e.date},${type},${e.category},${e.amount},${e.recorder},"${e.description || ''}"\n`;
    });
    const blob = new Blob(["\uFEFF" + csv], {type: "text/csv;charset=utf-8;"});
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = "รายรับรายจ่าย_ญส_" + new Date().toISOString().slice(0,10) + ".csv";
    link.click();
  }

  render();
</script>
